        function startQuiz() {
            currentQuestionIndex = 0;
            score = 0;
            userAnswers.fill(null);
            renderQuestion();
        }

        function renderQuestion() {
            const q = questions[currentQuestionIndex];
            const progress = ((currentQuestionIndex) / questions.length) * 100;

            let optionsHtml = '';
            q.options.forEach((opt, idx) => {
                const isSelected = userAnswers[currentQuestionIndex] === idx;
                const baseClass = "option-btn w-full text-left p-4 rounded-lg border-2 mb-3 text-gray-700 font-medium";
                const selectedClass = isSelected ? "border-blue-500 bg-blue-50" : "border-gray-200 bg-white hover:border-blue-300";
                
                optionsHtml += `
                    <button onclick="selectAnswer(${idx})" class="${baseClass} ${selectedClass}">
                        ${opt}
                    </button>
                `;
            });

            app.innerHTML = `
                <div class="bg-white rounded-xl shadow-lg p-8 fade-in">
                    <!-- Progress Bar -->
                    <div class="w-full bg-gray-200 rounded-full h-2.5 mb-6">
                        <div class="bg-blue-600 h-2.5 rounded-full transition-all duration-300" style="width: ${progress}%"></div>
                    </div>
                    
                    <div class="flex justify-between items-center mb-6">
                        <span class="text-sm font-bold text-gray-500 uppercase tracking-wider">Question ${currentQuestionIndex + 1} of ${questions.length}</span>
                        <div class="flex space-x-2">
                            <span class="bg-indigo-100 text-indigo-800 text-xs px-3 py-1 rounded-full font-semibold border border-indigo-200">${q.gottlieb}</span>
                            <span class="bg-emerald-100 text-emerald-800 text-xs px-3 py-1 rounded-full font-semibold border border-emerald-200">${q.lin}</span>
                        </div>
                    </div>

                    <div class="flex items-start mb-8">
                        <div class="text-4xl mr-4 bg-slate-100 p-3 rounded-xl">${q.icon}</div>
                        <h2 class="text-xl font-bold text-gray-800 leading-relaxed">${q.question}</h2>
                    </div>

                    <div class="mb-8">
                        ${optionsHtml}
                    </div>

                    <div class="flex justify-between items-center border-t pt-6">
                        <button onclick="prevQuestion()" class="px-6 py-2 rounded-lg font-semibold text-gray-600 hover:bg-gray-100 transition ${currentQuestionIndex === 0 ? 'invisible' : ''}">
                            &larr; Previous
                        </button>
                        
                        <button id="nextBtn" onclick="nextQuestion()" disabled class="px-8 py-3 rounded-lg font-bold text-white transition bg-gray-300 cursor-not-allowed">
                            ${currentQuestionIndex === questions.length - 1 ? 'Submit Assessment' : 'Next Question &rarr;'}
                        </button>
                    </div>
                </div>
            `;

            // Enable next button if an answer is already selected (when going back)
            if (userAnswers[currentQuestionIndex] !== null) {
                enableNextButton();
            }
        }

        function selectAnswer(idx) {
            userAnswers[currentQuestionIndex] = idx;
            renderQuestion(); // Re-render to show selected state
            enableNextButton();
        }

        function enableNextButton() {
            const btn = document.getElementById('nextBtn');
            btn.disabled = false;
            btn.classList.remove('bg-gray-300', 'cursor-not-allowed');
            if (currentQuestionIndex === questions.length - 1) {
                btn.classList.add('bg-green-600', 'hover:bg-green-700');
            } else {
                btn.classList.add('bg-blue-600', 'hover:bg-blue-700');
            }
        }

        function prevQuestion() {
            if (currentQuestionIndex > 0) {
                currentQuestionIndex--;
                renderQuestion();
            }
        }

        function nextQuestion() {
            if (currentQuestionIndex < questions.length - 1) {
                currentQuestionIndex++;
                renderQuestion();
            } else {
                calculateScore();
                renderResults();
            }
        }

        function calculateScore() {
            score = 0;
            userAnswers.forEach((ans, idx) => {
                if (ans === questions[idx].correct) {
                    score++;
                }
            });
        }

        function renderResults() {
            const percentage = Math.round((score / questions.length) * 100);
            
            // Academic Feedback Logic
            let feedbackTitle = "";
            let feedbackText = "";
            let feedbackColor = "";

            if (score >= 13) {
                feedbackTitle = "Outstanding Integration of Content & Language!";
                feedbackColor = "text-green-600";
                feedbackText = `
                    <p class="mb-2"><strong>Excellent work!</strong> You have demonstrated a strong command of both the scientific content regarding petroleum and the academic language required to express it.</p>
                    <p><strong>Language Feedback:</strong> Your ability to comprehend complex sentence structures (like cause/effect) and identify text-level genres (argumentative, expository) is outstanding. Keep using precise vocabulary like <em>"hydrocarbons"</em> and sophisticated transition words in your own scientific writing!</p>
                `;
            } else if (score >= 8) {
                feedbackTitle = "Good Progress in Scientific Literacy!";
                feedbackColor = "text-blue-600";
                feedbackText = `
                    <p class="mb-2"><strong>Good job!</strong> You have a solid grasp of the basic vocabulary (word level) and the core concepts of petroleum extraction and use.</p>
                    <p><strong>Language Feedback:</strong> To strengthen your academic language further, pay close attention to <em>Sentence Level</em> grammar. Notice how the passive voice is used to describe scientific processes (e.g., "is refined"), and practice building complex sentences that link causes to their environmental effects.</p>
                `;
            } else {
                feedbackTitle = "Let's Review Content & Language Strategies";
                feedbackColor = "text-amber-600";
                feedbackText = `
                    <p class="mb-2"><strong>Keep practicing!</strong> While you are picking up some of the content, we need to focus on strengthening your academic language to fully understand the science.</p>
                    <p><strong>Language Feedback:</strong> Let's work on recognizing key scientific vocabulary (Word Level) and understanding how sentences are built. Rereading texts and focusing on sequence words (e.g., <em>First, Subsequently</em>) will help you recount scientific processes more accurately next time.</p>
                `;
            }

            // Generate Review List
            let reviewHtml = '<div class="mt-8 space-y-4">';
            questions.forEach((q, idx) => {
                const isCorrect = userAnswers[idx] === q.correct;
                const icon = isCorrect ? '✅' : '❌';
                const bgColor = isCorrect ? 'bg-green-50 border-green-200' : 'bg-red-50 border-red-200';
                
                reviewHtml += `
                    <div class="p-4 border rounded-lg ${bgColor}">
                        <p class="font-semibold text-gray-800 text-sm mb-1">
                            ${icon} Q${idx + 1}: <span class="font-normal text-gray-600">[${q.gottlieb} / ${q.lin}]</span>
                        </p>
                        <p class="text-sm text-gray-700">${q.question}</p>
                        ${!isCorrect ? `<p class="text-sm text-red-600 mt-2"><strong>Correct Answer:</strong> ${q.options[q.correct]}</p>` : ''}
                    </div>
                `;
            });
            reviewHtml += '</div>';

            app.innerHTML = `
                <div class="bg-white rounded-xl shadow-lg p-8 fade-in border-t-8 ${score >= 8 ? (score >= 13 ? 'border-green-500' : 'border-blue-500') : 'border-amber-500'}">
                    <div class="text-center mb-8 border-b pb-8">
                        <h2 class="text-4xl font-black text-gray-800 mb-2">Your Score: ${score} / ${questions.length}</h2>
                        <p class="text-xl font-medium text-gray-500">${percentage}% Correct</p>
                    </div>

                    <div class="bg-slate-50 rounded-xl p-6 mb-8 border border-slate-200">
                        <h3 class="text-2xl font-bold ${feedbackColor} mb-3">${feedbackTitle}</h3>
                        <div class="text-gray-700 leading-relaxed">
                            ${feedbackText}
                        </div>
                    </div>

                    <div class="mb-8">
                        <button onclick="document.getElementById('reviewSection').classList.toggle('hidden')" class="w-full bg-slate-200 hover:bg-slate-300 text-slate-800 font-semibold py-3 px-6 rounded-lg transition duration-200">
                            Toggle Answer Review 📝
                        </button>
                        <div id="reviewSection" class="hidden">
                            ${reviewHtml}
                        </div>
                    </div>

                    <button onclick="renderIntro()" class="w-full bg-blue-600 hover:bg-blue-700 text-white font-bold py-4 px-6 rounded-lg transition duration-200 shadow-md text-lg">
                        Return to Start 🔄
                    </button>
                </div>
            `;
        }

        renderIntro();
